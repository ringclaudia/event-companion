Lead Database (Google Sheets)

Lets your team sync captured leads to a shared Google Sheet in real-time.

Setup instructions (one-time, ~2 min)
  1. Create a new Google Sheet
  2. Add these headers in Row 1: id | eventName | firstName | lastName | title | company | linkedin | email | phone | priority | source | notes | capturedBy | timestamp
  3. Go to Extensions > Apps Script
  4. Delete the default code and paste:
    function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length < 2) return ContentService.createTextOutput('[]').setMimeType(ContentService.MimeType.JSON);
  var headers = data[0];
  var rows = data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(h, i) { obj[h] = row[i]; });
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  var payload = JSON.parse(e.postData.contents);
  var leads = Array.isArray(payload) ? payload : [payload];
  leads.forEach(function(lead) {
    // Skip duplicates by id
    var ids = sheet.getRange(1, 1, sheet.getLastRow(), 1).getValues().flat();
    if (lead.id && ids.indexOf(lead.id) !== -1) return;
    var row = headers.map(function(h) { return lead[h] !== undefined ? lead[h] : ''; });
    sheet.appendRow(row);
  });
  return ContentService.createTextOutput(JSON.stringify({status:'ok',count:leads.length})).setMimeType(ContentService.MimeType.JSON);
}
  5. Click Deploy > New deployment
  6. Type = Web app, Execute as = Me, Who has access = Anyone
  7. Click Deploy, authorize when prompted, then copy the Web app URL below
